# Host (desktop) build of the JNI bridge, for the JVM integration test WhisperNativeHostTest only.
# Usage: scripts/host-whisper-jni/build.sh   (output: build/host-whisper-jni/)
cmake_minimum_required(VERSION 3.22)
project(voicememo_whisper_host C CXX)
find_package(JNI REQUIRED)
set(WHISPER_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/whisper.cpp)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
add_subdirectory(${WHISPER_DIR} whisper EXCLUDE_FROM_ALL)
include(${CMAKE_CURRENT_SOURCE_DIR}/../../app/src/main/cpp/whisper_patches.cmake)
add_library(vmwhisper SHARED ${CMAKE_CURRENT_SOURCE_DIR}/../../app/src/main/cpp/voicememo_whisper.cpp)
target_include_directories(vmwhisper PRIVATE ${JNI_INCLUDE_DIRS})
target_link_libraries(vmwhisper PRIVATE whisper ggml)
set_target_properties(vmwhisper PROPERTIES BUILD_RPATH "$ORIGIN")
